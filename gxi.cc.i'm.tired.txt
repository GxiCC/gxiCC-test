-- Dragging Helper
local function MakeDraggable(guiObject, handleObject)
    handleObject = handleObject or guiObject
    local dragging, dragInput, dragStart, startPos
    
    handleObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = guiObject.Position
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    
    handleObject.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            guiObject.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

    local Frame = Instance.new("Frame", column)
    Frame.Size = UDim2.new(0.94, 0, 0, 34)
    Frame.BackgroundColor3 = Color3.fromRGB(16, 25, 38)
    Frame.BackgroundTransparency = 0.4
    Frame.LayoutOrder = layoutOrder

    local Corner = Instance.new("UICorner", Frame)

    local Label = Instance.new("TextLabel", Frame)
    Label.Size = UDim2.new(1, 0, 0, 14)
    Label.BackgroundTransparency = 1
    Label.Text = text .. ": " .. tostring(Config[key])
    Label.TextColor3 = Color3.fromRGB(200, 215, 230)
    Label.TextSize = 10
    Label.Font = Enum.Font.SourceSans

    local SliderBtn = Instance.new("TextButton", Frame)
    SliderBtn.Size = UDim2.new(0.9, 0, 0, 4)
    SliderBtn.Position = UDim2.new(0.05, 0, 0.65, 0)
    SliderBtn.BackgroundColor3 = Color3.fromRGB(30, 45, 65)
    SliderBtn.Text = ""

    local SliderCorner = Instance.new("UICorner", SliderBtn)
    SliderCorner.CornerRadius = UDim.new(1, 0)

    local Fill = Instance.new("Frame", SliderBtn)
    Fill.Size = UDim2.new((Config[key] - min)/(max - min), 0, 1, 0)
    Fill.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    Fill.BorderSizePixel = 0

    local FillCorner = Instance.new("UICorner", Fill)
    FillCorner.CornerRadius = UDim.new(1, 0)

    local dragging = false
    local function Update(input)
        local pos = math.clamp((input.Position.X - SliderBtn.AbsolutePosition.X) / SliderBtn.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + ((max - min) * pos))
        Fill.Size = UDim2.new(pos, 0, 1, 0)
        Label.Text = text .. ": " .. tostring(val)
        Config[key] = val
        if callback then callback(val) end
    end

    SliderBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            Update(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            Update(input)
        end
    end)
end

-- TEAM COLOR LOOKUP (Prison Life Accurate)
local function GetTeamColor(player)
    local team = tostring(player.Team)
    if team == "Guards" then
        return Color3.fromRGB(0, 120, 255) -- Blue
    elseif team == "Inmates" then
        return Color3.fromRGB(255, 130, 0) -- Orange
    elseif team == "Criminals" then
        return Color3.fromRGB(255, 30, 30) -- Red
    end
    return player.TeamColor and player.TeamColor.Color or Color3.fromRGB(200, 200, 200)
end

-- TARGETING & ACCURATE TEAM CHECK
local function IsValidTarget(player)
    if not player or player == LocalPlayer or not player.Character then return false end
    
    local hum = player.Character:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end

    local myTeam = tostring(LocalPlayer.Team)
    local targetTeam = tostring(player.Team)

    if Config.TeamCheck then
        -- Ignore teammates unconditionally
        if myTeam == targetTeam and myTeam ~= "Neutral" then return false end
        
        if Config.TargetTeam == "Guards" and targetTeam ~= "Guards" then return false end
        if Config.TargetTeam == "Inmates" and targetTeam ~= "Inmates" then return false end
        if Config.TargetTeam == "Criminals" and targetTeam ~= "Criminals" then return false end
    end

    -- WallCheck Raycast
    if Config.WallCheck then
        local targetPart = player.Character:FindFirstChild(Config.AimTarget) or player.Character:FindFirstChild("Head")
        if targetPart then
            local rayParams = RaycastParams.new()
            rayParams.FilterType = Enum.RaycastFilterType.Exclude
            rayParams.FilterDescendantsInstances = {LocalPlayer.Character, player.Character}
            
            local origin = Camera.CFrame.Position
            local direction = targetPart.Position - origin
            local rayResult = Workspace:Raycast(origin, direction, rayParams)

            if rayResult then return false end -- Occluded by wall
        end
    end

    return true
end

local function GetFOVCenter()
    if Config.FOV_Mode == "Static" then
        return Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    else
        return UserInputService:GetMouseLocation()
    end
end

local function GetClosestPlayer()
    local closestPart = nil
    local shortestDistance = Config.FOV_Size
    local center = GetFOVCenter()

    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) and player.Character:FindFirstChild(Config.AimTarget) then
            local part = player.Character[Config.AimTarget]
            local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)

            if onScreen then
                local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                if dist < shortestDistance then
                    shortestDistance = dist
                    closestPart = part
                end
            end
        end
    end
    return closestPart
end

local function CreateESPComponents(player)
    if player == LocalPlayer then return end

    local function CharacterAdded(char)
        if not char then return end

        local hl = Instance.new("Highlight")
        hl.Name = "GxiHighlightNL"
        hl.Parent = char
        hl.FillColor = GetTeamColor(player)
        hl.OutlineColor = Color3.fromRGB(255, 255, 255)
        hl.FillTransparency = 0.5
        hl.Enabled = Config.ESP

        task.spawn(function()
            local head = char:WaitForChild("Head", 5)
            if head then
                local bbg = Instance.new("BillboardGui")
                bbg.Name = "GxiBillboardNL"
                bbg.Parent = head
                bbg.Adornee = head
                bbg.Size = UDim2.new(0, 140, 0, 45)
                bbg.StudsOffset = Vector3.new(0, 2.3, 0)
                bbg.AlwaysOnTop = true

                local nameLabel = Instance.new("TextLabel", bbg)
                nameLabel.Name = "NameLabel"
                nameLabel.Size = UDim2.new(1, 0, 0, 18)
                nameLabel.BackgroundTransparency = 1
                nameLabel.Text = player.DisplayName
                nameLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
                nameLabel.TextSize = Config.NameSize
                nameLabel.Font = Enum.Font.SourceSansBold
                nameLabel.TextStrokeTransparency = 0.4
                nameLabel.Visible = Config.Names

                local hpBg = Instance.new("Frame", bbg)
                hpBg.Name = "HPBg"
                hpBg.Size = UDim2.new(0, 70, 0, 3)
                hpBg.Position = UDim2.new(0.5, -35, 0, 20)
                hpBg.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
                hpBg.BorderSizePixel = 0
                hpBg.Visible = Config.HPBar

                local hpFill = Instance.new("Frame", hpBg)
                hpFill.Name = "HPFill"
                hpFill.Size = UDim2.new(1, 0, 1, 0)
                hpFill.BackgroundColor3 = Color3.fromRGB(0, 255, 120)
                hpFill.BorderSizePixel = 0
            end
        end)

        for _, part in ipairs(char:GetChildren()) do
            if part:IsA("BasePart") then
                local box = Instance.new("BoxHandleAdornment")
                box.Name = "GxiChamsNL"
                box.Parent = part
                box.Adornee = part
                box.Size = part.Size
                box.Color3 = GetTeamColor(player)
                box.Transparency = 0.4
                box.AlwaysOnTop = true
                box.ZIndex = 10
                box.Visible = Config.Chams
            end
        end
    end

    if player.Character then CharacterAdded(player.Character) end
    player.CharacterAdded:Connect(CharacterAdded)
end

-- MAIN RENDER LOOP
RunService.RenderStepped:Connect(function(deltaTime)
    -- FOV Circle Update
    if FOVCircle then
        FOVCircle.Visible = Config.FOV_Visible
        FOVCircle.Radius = Config.FOV_Size
        FOVCircle.Color = Config.FOV_Color
        FOVCircle.Position = GetFOVCenter()
    end

    -- Visuals Update Loop
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local char = player.Character
            local teamCol = GetTeamColor(player)
            local activeColor = Config.TeamColorChams and teamCol or Config.ESP_Color

            local hl = char:FindFirstChild("GxiHighlightNL")
            if hl then
                hl.Enabled = Config.ESP
                hl.FillColor = activeColor
            end

            for _, child in ipairs(char:GetChildren()) do
                if child:IsA("BoxHandleAdornment") and child.Name == "GxiChamsNL" then
                    child.Visible = Config.Chams
                    child.Color3 = activeColor
                end
            end

            local head = char:FindFirstChild("Head")
            if head then
                local bbg = head:FindFirstChild("GxiBillboardNL")
                if bbg then
                    local nameLabel = bbg:FindFirstChild("NameLabel")
                    if nameLabel then
                        nameLabel.Visible = Config.Names
                        nameLabel.TextSize = Config.NameSize
                    end

                    local hpBg = bbg:FindFirstChild("HPBg")
                    if hpBg then
                        hpBg.Visible = Config.HPBar
                        local hpFill = hpBg:FindFirstChild("HPFill")
                        local hum = char:FindFirstChildOfClass("Humanoid")
                        if hpFill and hum then
                            local healthPct = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
                            hpFill.Size = UDim2.new(healthPct, 0, 1, 0)
                        end
                    end
                end
            end
        end
    end

    -- SEPARATED AIMBOT & SILENT AIM
    local targetPart = GetClosestPlayer()
    local isShooting = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)

    if targetPart then
        if Config.Aimbot then
            -- Smooth Camera Aim
            local targetCamPos = CFrame.new(Camera.CFrame.Position, targetPart.Position)
            local smoothnessFactor = math.clamp((Config.AimSpeed / 100) * (deltaTime * 60), 0.01, 1)
            Camera.CFrame = Camera.CFrame:Lerp(targetCamPos, smoothnessFactor)
        elseif Config.SilentAim and isShooting then
            -- Instant Firing Snap
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, targetPart.Position)
        end
    end
end)
